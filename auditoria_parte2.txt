        public AuthService(
            IUnitOfWork uow,
            IConfiguration config,

            _uow = uow;
            _config = config;

        public async Task<ApiResponse<AuthResponseDto>> LoginAsync(
            LoginRequestDto dto,
            string ip = "127.0.0.1",
            string dispositivo = "WebClient")
        {
            var identificador = dto.UsernameOrEmail.Trim();
            var usuario = await _uow.Usuarios.GetByUsernameOrEmailAsync(identificador);

            if (usuario == null)
                return ApiResponse<AuthResponseDto>.Falla("Usuario o contraseña incorrectos.");

            if (!usuario.Activo)
                return ApiResponse<AuthResponseDto>.Falla("La cuenta de usuario se encuentra desactivada.");

            if (usuario.BloqueadoHasta.HasValue && usuario.BloqueadoHasta.Value > DateTime.UtcNow)
            {
                await RegistrarIntentoSesionAsync(usuario.Id, ip, dispositivo, false, MotivoFalloSesionEnum.CuentaBloqueada);
                return ApiResponse<AuthResponseDto>.Falla($"La cuenta se encuentra bloqueada temporalmente hasta {usuario.BloqueadoHasta.Value:HH:mm:ss} UTC.");
            }

            bool passwordValida = BCrypt.Net.BCrypt.Verify(dto.Password, usuario.PasswordHash);
            if (!passwordValida)
            {
                usuario.IntentosFallidos += 1;
                if (usuario.IntentosFallidos >= 5)
                {
                    usuario.BloqueadoHasta = DateTime.UtcNow.AddMinutes(15);
                }

                _uow.Usuarios.Update(usuario);
                await RegistrarIntentoSesionAsync(usuario.Id, ip, dispositivo, false, MotivoFalloSesionEnum.CredencialesIncorrectas);

                return ApiResponse<AuthResponseDto>.Falla("Usuario o contraseña incorrectos.");
            }

            // Login Exitoso: restablecer contador de fallos y registrar sesión
            usuario.IntentosFallidos = 0;
            usuario.BloqueadoHasta = null;
            _uow.Usuarios.Update(usuario);

            await RegistrarIntentoSesionAsync(usuario.Id, ip, dispositivo, true, MotivoFalloSesionEnum.Ninguno);

            int tenantId = _tenantProvider.CurrentTenantId > 0 ? _tenantProvider.CurrentTenantId : 1;
            int sucursalId = _tenantProvider.CurrentSucursalId ?? 1;

            var jwtKey = _config["Jwt:Key"] ?? "ClaveSecretaSuperSeguraMiPeluqueriaSaaS2026!";
            var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtKey));
            var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);
            var rolNombre = usuario.Rol?.Nombre ?? "Cliente";
            var expiracionJwt = DateTime.UtcNow.AddHours(8);

            var claims = new[]
            {
                new Claim(JwtRegisteredClaimNames.Sub, usuario.Id.ToString()),
                new Claim(ClaimTypes.NameIdentifier, usuario.Id.ToString()),
                new Claim(JwtRegisteredClaimNames.Email, usuario.Email),
                new Claim("Username", usuario.Username),
                new Claim(ClaimTypes.Role, rolNombre),
                new Claim("TenantId", tenantId.ToString()),
                new Claim("SucursalId", sucursalId.ToString())
            };

            var token = new JwtSecurityToken(
                issuer: _config["Jwt:Issuer"],
                audience: _config["Jwt:Audience"],
                claims: claims,
                expires: expiracionJwt,
                signingCredentials: creds
            );

            var tokenString = new JwtSecurityTokenHandler().WriteToken(token);

            // Generar y persistir RefreshToken en base de datos
            var refreshTokenString = Convert.ToBase64String(RandomNumberGenerator.GetBytes(64));
            var refreshTokenEntity = new RefreshToken
            {
                UsuarioId = usuario.Id,
                Token = refreshTokenString,
                ExpiresAt = DateTime.UtcNow.AddDays(7),
                CreatedByIp = ip,
                CreatedByUserId = usuario.Id
            };

            await _uow.Usuarios.AddRefreshTokenAsync(refreshTokenEntity);
            await _uow.SaveChangesAsync();

            var response = new AuthResponseDto
            {
                UsuarioId = usuario.Id,
                Username = usuario.Username,
                Email = usuario.Email,
                Rol = rolNombre,
                Token = tokenString,
                RefreshToken = refreshTokenString,
                Expiracion = expiracionJwt,
                TenantId = tenantId,
                SucursalId = sucursalId
            };

            return ApiResponse<AuthResponseDto>.Exito(response, "Inicio de sesión exitoso.");
        }

        public async Task<ApiResponse<UsuarioResponseDto>> RegistrarUsuarioAsync(RegistroUsuarioDto dto)
        {
            var emailNorm = dto.Email.Trim().ToLower();
            var userNorm = dto.Username.Trim();

            bool existe = await _uow.Usuarios.ExistsUsernameOrEmailAsync(userNorm, emailNorm);
            if (existe)
                return ApiResponse<UsuarioResponseDto>.Falla("El nombre de usuario o correo electrónico ya se encuentra registrado.");

            string codigoOtp = Random.Shared.Next(100000, 999999).ToString();

            var nuevoUsuario = new Usuario
            {
                Username = userNorm,
                Email = emailNorm,
                PasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.Password),
                RolId = dto.RolId > 0 ? dto.RolId : 4,
                IntentosFallidos = 0,
                EmailVerificado = false,
                CodigoVerificacion = codigoOtp,
                FechaExpiracionCodigo = DateTime.UtcNow.AddMinutes(15),
                AceptaTerminos = dto.AceptaTerminos,
                FechaAceptacionTerminos = DateTime.UtcNow,
                Activo = true,
                ClienteId = dto.ClienteId,
                PeluqueroId = dto.PeluqueroId
            };

            await _uow.Usuarios.AddAsync(nuevoUsuario);
            await _uow.SaveChangesAsync();

            await _emailService.EnviarCodigoOtpAsync(nuevoUsuario.Email, nuevoUsuario.Username, codigoOtp, "Verificación de Cuenta");

            var creado = await _uow.Usuarios.GetByIdWithRolAsync(nuevoUsuario.Id);

            return ApiResponse<UsuarioResponseDto>.Exito(
                MapToUsuarioDto(creado ?? nuevoUsuario),
                "Usuario registrado correctamente. Se envió un código de 6 dígitos a su correo electrónico.");
        }

        public async Task<ApiResponse<bool>> VerificarEmailAsync(VerificarEmailDto dto)
        {
            var usuario = await _uow.Usuarios.GetByEmailAsync(dto.Email);
            if (usuario == null)
                return ApiResponse<bool>.Falla("No existe un usuario registrado con ese correo electrónico.");

            if (string.IsNullOrWhiteSpace(usuario.CodigoVerificacion) || usuario.CodigoVerificacion != dto.CodigoVerificacion.Trim())
                return ApiResponse<bool>.Falla("El código de verificación ingresado es inválido.");

            if (usuario.FechaExpiracionCodigo.HasValue && usuario.FechaExpiracionCodigo.Value < DateTime.UtcNow)
                return ApiResponse<bool>.Falla("El código de verificación ha expirado.");

            usuario.EmailVerificado = true;
            usuario.CodigoVerificacion = null;
            usuario.FechaExpiracionCodigo = null;

            _uow.Usuarios.Update(usuario);
            await _uow.SaveChangesAsync();

            return ApiResponse<bool>.Exito(true, "Correo electrónico verificado exitosamente.");
        }

        public async Task<ApiResponse<bool>> CambiarPasswordAsync(int usuarioId, CambiarPasswordDto dto)
        {
            var usuario = await _uow.Usuarios.GetByIdWithRolAsync(usuarioId);
            if (usuario == null)
                return ApiResponse<bool>.Falla("El usuario no existe.");

            if (!BCrypt.Net.BCrypt.Verify(dto.PasswordActual, usuario.PasswordHash))
                return ApiResponse<bool>.Falla("La contraseña actual ingresada es incorrecta.");

            usuario.PasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.NuevaPassword);
            _uow.Usuarios.Update(usuario);
            await _uow.Usuarios.RevocarRefreshTokensUsuarioAsync(usuario.Id, "Cambio de contraseña por el usuario");
            await _uow.SaveChangesAsync();

            return ApiResponse<bool>.Exito(true, "Contraseña actualizada exitosamente.");
        }

        public async Task<ApiResponse<bool>> SolicitarRecuperacionPasswordAsync(OlvidePasswordDto dto)
        {
            var usuario = await _uow.Usuarios.GetByEmailAsync(dto.Email);
            if (usuario == null)
                return ApiResponse<bool>.Exito(true, "Si el correo existe en nuestro sistema, recibirá un código de recuperación en instantes.");

            string codigoOtp = Random.Shared.Next(100000, 999999).ToString();
            usuario.CodigoVerificacion = codigoOtp;
            usuario.FechaExpiracionCodigo = DateTime.UtcNow.AddMinutes(15);

            _uow.Usuarios.Update(usuario);
            await _uow.SaveChangesAsync();

            await _emailService.EnviarCodigoOtpAsync(usuario.Email, usuario.Username, codigoOtp, "Recuperación de Contraseña");

            return ApiResponse<bool>.Exito(true, "Se ha enviado un código OTP de 6 dígitos a su correo electrónico.");
        }

        public async Task<ApiResponse<bool>> RestablecerPasswordConOtpAsync(RestablecerPasswordDto dto)
        {
            var usuario = await _uow.Usuarios.GetByEmailAsync(dto.Email);
            if (usuario == null)
                return ApiResponse<bool>.Falla("Datos de recuperación inválidos.");

            if (string.IsNullOrWhiteSpace(usuario.CodigoVerificacion) || usuario.CodigoVerificacion != dto.CodigoOtp.Trim())
                return ApiResponse<bool>.Falla("El código OTP ingresado no es válido.");

            if (usuario.FechaExpiracionCodigo.HasValue && usuario.FechaExpiracionCodigo.Value < DateTime.UtcNow)
                return ApiResponse<bool>.Falla("El código OTP ha expirado. Solicite uno nuevo.");

            usuario.PasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.NuevaPassword);
            usuario.IntentosFallidos = 0;
            usuario.BloqueadoHasta = null;
            usuario.CodigoVerificacion = null;
            usuario.FechaExpiracionCodigo = null;

            _uow.Usuarios.Update(usuario);
            await _uow.Usuarios.RevocarRefreshTokensUsuarioAsync(usuario.Id, "Restablecimiento de contraseña vía OTP");
            await _uow.SaveChangesAsync();

            return ApiResponse<bool>.Exito(true, "Su contraseña ha sido restablecida exitosamente. Ya puede iniciar sesión.");
        }

        public async Task<ApiResponse<IEnumerable<UsuarioResponseDto>>> ObtenerUsuariosAsync()
        {
            var usuarios = await _uow.Usuarios.GetAllWithRolesAsync();
            return ApiResponse<IEnumerable<UsuarioResponseDto>>.Exito(usuarios.Select(MapToUsuarioDto));
        }

        public async Task<ApiResponse<UsuarioResponseDto>> ActualizarEstadoUsuarioAsync(int id, ActualizarEstadoUsuarioDto dto)
        {
            var usuario = await _uow.Usuarios.GetByIdWithRolAsync(id);
            if (usuario == null)
                return ApiResponse<UsuarioResponseDto>.Falla("El usuario solicitado no existe.");

            if (dto.RolId.HasValue && dto.RolId.Value > 0)
                usuario.RolId = dto.RolId.Value;

            if (dto.Activo.HasValue)
            {
                usuario.Activo = dto.Activo.Value;
                if (!dto.Activo.Value)
                {
                    await _uow.Usuarios.RevocarRefreshTokensUsuarioAsync(usuario.Id, "Usuario desactivado por administración");
                }
            }

            if (dto.DesbloquearCuenta)
            {
                usuario.IntentosFallidos = 0;
                usuario.BloqueadoHasta = null;
            }

            _uow.Usuarios.Update(usuario);
            await _uow.SaveChangesAsync();

            var actualizado = await _uow.Usuarios.GetByIdWithRolAsync(id);
            return ApiResponse<UsuarioResponseDto>.Exito(MapToUsuarioDto(actualizado ?? usuario), "Usuario actualizado correctamente.");
        }

        public async Task<ApiResponse<IEnumerable<HistorialSesionResponseDto>>> ObtenerHistorialSesionesAsync(int? usuarioId = null, int top = 100)
        {
            var registros = await _uow.Usuarios.GetHistorialSesionesAsync(usuarioId, top);

            var dtos = registros.Select(h => new HistorialSesionResponseDto
            {
                Id = h.Id,
                UsuarioId = h.UsuarioId,
                Username = h.Usuario?.Username ?? $"Usuario #{h.UsuarioId}",
                FechaHora = h.FechaIntento,
                IpOrigen = h.DireccionIP,
                Dispositivo = h.Dispositivo,
                Exitoso = h.Exitoso,
                MotivoFallo = h.MotivoFallo?.ToString()
            }).ToList();

            return ApiResponse<IEnumerable<HistorialSesionResponseDto>>.Exito(dtos);
        }

        private async Task RegistrarIntentoSesionAsync(
            int usuarioId,
            string ip,
            string dispositivo,
            bool exitoso,
            MotivoFalloSesionEnum motivo)
        {
            var sesion = new HistorialSesion
            {
                UsuarioId = usuarioId,
                DireccionIP = ip,
                Dispositivo = dispositivo,
                Exitoso = exitoso,
                MotivoFallo = motivo,
                FechaIntento = DateTime.UtcNow
            };

            await _uow.Usuarios.RegistrarHistorialSesionAsync(sesion);
            await _uow.SaveChangesAsync();
        }

        private static UsuarioResponseDto MapToUsuarioDto(Usuario u)
        {
            return new UsuarioResponseDto
            {
                Id = u.Id,
                Username = u.Username,
                Email = u.Email,
                RolId = u.RolId,
                RolNombre = u.Rol?.Nombre ?? string.Empty,
                EmailVerificado = u.EmailVerificado,
                Activo = u.Activo && !u.IsDeleted,
                IntentosFallidos = u.IntentosFallidos,
                BloqueadoHasta = u.BloqueadoHasta,
                ClienteId = u.ClienteId,
                PeluqueroId = u.PeluqueroId
            };
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using AutoMapper;
using Microsoft.EntityFrameworkCore;
using MiPeluqueria.Api.Data.Context;
using MiPeluqueria.Api.DTOs.Common;
using MiPeluqueria.Api.DTOs.Turnos;
using MiPeluqueria.Api.Models.Turnos;
using MiPeluqueria.Api.Services.Interfaces;

namespace MiPeluqueria.Api.Services.Implementations
{
    public class BloqueoHorarioService : IBloqueoHorarioService
    {
        private readonly ApplicationDbContext _context;
        private readonly IMapper _mapper;

        public BloqueoHorarioService(ApplicationDbContext context, IMapper mapper)
        {
            _context = context;
            _mapper = mapper;
        }

        public async Task<ApiResponse<IEnumerable<object>>> ObtenerPorPeluqueroAsync(int peluqueroId)
        {
            var bloqueos = await _context.BloqueosHorario
                .Where(b => b.PeluqueroId == peluqueroId)
                .OrderByDescending(b => b.Id)
                .AsNoTracking()
                .ToListAsync();

            var resultado = bloqueos.Select(b => (object)b).ToList();
            return ApiResponse<IEnumerable<object>>.Exito(resultado);
        }

        public async Task<ApiResponse<int>> CrearBloqueoAsync(BloqueoHorarioDto dto)
        {
            var peluquero = await _context.Peluqueros.FindAsync(dto.PeluqueroId);
            if (peluquero == null)
                return ApiResponse<int>.Falla("El peluquero especificado no existe.");

            var bloqueo = _mapper.Map<BloqueoHorario>(dto);
            await _context.BloqueosHorario.AddAsync(bloqueo);
            await _context.SaveChangesAsync();

            return ApiResponse<int>.Exito(bloqueo.Id, "Bloqueo horario registrado correctamente.");
        }

        public async Task<ApiResponse<bool>> EliminarBloqueoAsync(int id)
        {
            var bloqueo = await _context.BloqueosHorario.FindAsync(id);
            if (bloqueo == null)
                return ApiResponse<bool>.Falla("El bloqueo horario especificado no existe.");

            _context.BloqueosHorario.Remove(bloqueo);
            await _context.SaveChangesAsync();

            return ApiResponse<bool>.Exito(true, "Bloqueo horario eliminado correctamente.");
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using MiPeluqueria.Api.DTOs.Common;
using MiPeluqueria.Api.DTOs.Ventas;
using MiPeluqueria.Api.Models.Ventas;
using MiPeluqueria.Api.Repositories.Interfaces;
using MiPeluqueria.Api.Services.Interfaces;

namespace MiPeluqueria.Api.Services.Implementations
{
    public class CajaService : ICajaService
    {
        private readonly IUnitOfWork _unitOfWork;

        public CajaService(IUnitOfWork unitOfWork)
        {
            _unitOfWork = unitOfWork;
        }

        public async Task<ApiResponse<SesionCaja?>> ObtenerEstadoCajaActualAsync(CancellationToken ct = default)
        {
            var caja = await _unitOfWork.SesionesCaja.GetCajaAbiertaActualAsync(ct);
            return ApiResponse<SesionCaja?>.Ok(caja, caja != null ? "Caja abierta encontrada." : "No hay ninguna caja abierta actualmente.");
        }

        public async Task<ApiResponse<IReadOnlyList<SesionCaja>>> ObtenerHistorialCajasAsync(DateTime? desde = null, DateTime? hasta = null, CancellationToken ct = default)
        {
            var historial = await _unitOfWork.SesionesCaja.GetHistorialCajasAsync(desde, hasta, ct);
            return ApiResponse<IReadOnlyList<SesionCaja>>.Ok(historial);
        }

        public async Task<ApiResponse<SesionCaja>> AbrirCajaAsync(AperturaCajaDto dto, CancellationToken ct = default)
        {
            var cajaAbierta = await _unitOfWork.SesionesCaja.GetCajaAbiertaActualAsync(ct);
            if (cajaAbierta != null)
                return ApiResponse<SesionCaja>.Fail($"Ya existe una sesión de caja abierta (ID #{cajaAbierta.Id}). Debe cerrarla antes de abrir una nueva.");

            var nuevaSesion = new SesionCaja
            {
                UsuarioAperturaId = dto.UsuarioId,
                FechaApertura = DateTime.UtcNow,
                MontoInicial = dto.MontoInicial,
                EstadoAbierta = true
            };

            await _unitOfWork.SesionesCaja.AbrirSesionCajaAsync(nuevaSesion, ct);
            await _unitOfWork.SaveChangesAsync(ct);

            return ApiResponse<SesionCaja>.Ok(nuevaSesion, "Caja abierta exitosamente.");
        }

        public async Task<ApiResponse<MovimientoCaja>> RegistrarIngresoCajaAsync(int sesionCajaId, decimal monto, string concepto, string tipo = "IngresoManual", CancellationToken ct = default)
        {
            var caja = await _unitOfWork.SesionesCaja.GetCajaAbiertaActualAsync(ct);
            if (caja == null || (sesionCajaId > 0 && caja.Id != sesionCajaId))
                return ApiResponse<MovimientoCaja>.Fail("No hay una sesión de caja abierta válida para registrar el ingreso.");

            var movimiento = new MovimientoCaja
            {
                SesionCajaId = caja.Id,
                Tipo = string.IsNullOrWhiteSpace(tipo) ? "IngresoManual" : tipo.Trim(),
                Monto = monto,
                Concepto = concepto.Trim(),
                Fecha = DateTime.UtcNow
            };

            await _unitOfWork.SesionesCaja.RegistrarMovimientoCajaAsync(movimiento, ct);
            await _unitOfWork.SaveChangesAsync(ct);

            return ApiResponse<MovimientoCaja>.Ok(movimiento, "Ingreso manual registrado correctamente.");
        }

        public async Task<ApiResponse<MovimientoCaja>> RegistrarEgresoCajaAsync(int sesionCajaId, decimal monto, string concepto, string tipo = "EgresoMenor", CancellationToken ct = default)
        {
            var caja = await _unitOfWork.SesionesCaja.GetCajaAbiertaActualAsync(ct);
            if (caja == null || (sesionCajaId > 0 && caja.Id != sesionCajaId))
                return ApiResponse<MovimientoCaja>.Fail("No hay una sesión de caja abierta válida para registrar el egreso.");

            var movimiento = new MovimientoCaja
            {
                SesionCajaId = caja.Id,
                Tipo = string.IsNullOrWhiteSpace(tipo) ? "EgresoMenor" : tipo.Trim(),
                Monto = monto,
                Concepto = concepto.Trim(),
                Fecha = DateTime.UtcNow
            };

            await _unitOfWork.SesionesCaja.RegistrarMovimientoCajaAsync(movimiento, ct);
            await _unitOfWork.SaveChangesAsync(ct);

            return ApiResponse<MovimientoCaja>.Ok(movimiento, "Egreso de caja registrado correctamente.");
        }

        public async Task<ApiResponse<SesionCaja>> CerrarCajaAsync(CierreCajaDto dto, CancellationToken ct = default)
        {
            var caja = await _unitOfWork.SesionesCaja.GetCajaAbiertaActualAsync(ct);
            if (caja == null)
                return ApiResponse<SesionCaja>.Fail("No existe ninguna sesión de caja abierta para cerrar.");

            decimal totalVentas = caja.Ventas.Where(v => !v.IsDeleted).Sum(v => v.Total);
            decimal totalIngresosManuales = caja.MovimientosCaja
                .Where(m => !m.IsDeleted && m.Tipo.StartsWith("Ingreso", StringComparison.OrdinalIgnoreCase))
                .Sum(m => m.Monto);
            decimal totalEgresos = caja.MovimientosCaja
                .Where(m => !m.IsDeleted && !m.Tipo.StartsWith("Ingreso", StringComparison.OrdinalIgnoreCase))
                .Sum(m => m.Monto);

            decimal esperado = caja.MontoInicial + totalVentas + totalIngresosManuales - totalEgresos;

            caja.UsuarioCierreId = dto.UsuarioId;
            caja.FechaCierre = DateTime.UtcNow;
            caja.MontoFinalReal = dto.MontoFinalReal;
            caja.MontoFinalEsperado = esperado;
            caja.Diferencia = dto.MontoFinalReal - esperado;
            caja.EstadoAbierta = false;

            _unitOfWork.SesionesCaja.ActualizarSesionCaja(caja);
            await _unitOfWork.SaveChangesAsync(ct);

            return ApiResponse<SesionCaja>.Ok(caja, "Caja cerrada y arqueada exitosamente.");
        }
    }
}

        public async Task<ApiResponse<CatalogosMaestrosResponseDto>> ObtenerTodosLosCatalogosAsync()

            var response = new CatalogosMaestrosResponseDto

                         ?? type.GetProperty("Descripcion")?.GetValue(entity)?.ToString()
                         ?? string.Empty;
            var descripcion = type.GetProperty("Descripcion")?.GetValue(entity)?.ToString();
            var color = type.GetProperty("ColorHex")?.GetValue(entity)?.ToString();

            return new CatalogoItemDto
            {

        {
            var type = entity.GetType();
            var propNombre = type.GetProperty("Nombre");
            var propDescripcion = type.GetProperty("Descripcion");

            if (propNombre != null && propNombre.CanWrite)
                propNombre.SetValue(entity, nombre);

            if (propDescripcion != null && propDescripcion.CanWrite)
                propDescripcion.SetValue(entity, descripcion ?? nombre);
        }

        public async Task<ApiResponse<PagedResult<ClienteResponseDto>>> ObtenerTodosAsync(
            int pageNumber = 1,
            int pageSize = 10,
            string? search = null)
        {
            if (pageNumber < 1) pageNumber = 1;
            if (pageSize < 1) pageSize = 10;
            if (pageSize > 100) pageSize = 100;

            var query = _repository.GetAllQueryable();

            if (!string.IsNullOrWhiteSpace(search))
            {
                var term = search.Trim().ToLower();
                query = query.Where(c =>
                    c.Nombre.ToLower().Contains(term) ||
                    c.Apellido.ToLower().Contains(term) ||
                    c.Telefono.Contains(term) ||
                    (c.Email != null && c.Email.ToLower().Contains(term)));
            }

            int totalRecords = await query.CountAsync();

            var clientesPagina = await query
                .OrderBy(c => c.Apellido)
                .ThenBy(c => c.Nombre)
                .Skip((pageNumber - 1) * pageSize)
                .Take(pageSize)
                .ToListAsync();

            var dtos = _mapper.Map<List<ClienteResponseDto>>(clientesPagina);

            var mesActual = DateTime.UtcNow.Month;
            foreach (var c in dtos)
            {
                c.EsCumpleaneroDelMes = (c.FechaNacimiento.Month == mesActual);
            }

            var pagedResult = new PagedResult<ClienteResponseDto>
            {
                Data = dtos,
                TotalRecords = totalRecords,
                PageNumber = pageNumber,
                PageSize = pageSize
            };

            return ApiResponse<PagedResult<ClienteResponseDto>>.Exito(pagedResult, "Clientes obtenidos con éxito.");
        }

        public async Task<ApiResponse<ClienteResponseDto>> ObtenerPorIdAsync(int id)

            var dto = _mapper.Map<ClienteResponseDto>(cliente);
            dto.EsCumpleaneroDelMes = (dto.FechaNacimiento.Month == DateTime.UtcNow.Month);

            return ApiResponse<ClienteResponseDto>.Exito(dto, "Cliente obtenido con éxito.");
        }

                return ApiResponse<ClienteResponseDto>.Falla("Ya existe un cliente registrado con ese correo electrónico.");

            var cliente = _mapper.Map<Cliente>(dto);

            await _repository.AddAsync(cliente);
            await _unitOfWork.SaveChangesAsync();

            var creado = await _repository.GetByIdAsync(cliente.Id);

            if (!string.IsNullOrWhiteSpace(dto.Email) &&

        {
            var peluquero = await _context.Peluqueros.FindAsync(dto.PeluqueroId);
            if (peluquero == null)
                return ApiResponse<ComisionConfigDto>.Fail("El peluquero especificado no existe.");

            if (dto.ServicioId.HasValue)
            {
                var servicio = await _context.Servicios.FindAsync(dto.ServicioId.Value);
                if (servicio == null)
                    return ApiResponse<ComisionConfigDto>.Fail("El servicio especificado no existe.");
            }

            // Si ya existe una regla para ese peluquero y ese mismo ServicioId (o null para regla general), la actualiza (Upsert)
            var existente = await _context.ComisionesConfigs
                .FirstOrDefaultAsync(c => c.PeluqueroId == dto.PeluqueroId && c.ServicioId == dto.ServicioId);

            if (existente != null)
            {
                existente.PorcentajeServicio = dto.PorcentajeServicio;
                existente.PorcentajeVentaProducto = dto.PorcentajeVentaProducto;
                await _context.SaveChangesAsync();

                await _context.Entry(existente).Reference(c => c.Peluquero).LoadAsync();
                if (existente.ServicioId.HasValue)
                    await _context.Entry(existente).Reference(c => c.Servicio).LoadAsync();

                return ApiResponse<ComisionConfigDto>.Ok(MapToDto(existente), "Esquema de comisión actualizado correctamente.");
            }

            var nuevaConfig = new ComisionesConfig
            {
                PeluqueroId = dto.PeluqueroId,
                ServicioId = dto.ServicioId,
                PorcentajeServicio = dto.PorcentajeServicio,
                PorcentajeVentaProducto = dto.PorcentajeVentaProducto
            };

            await _context.SaveChangesAsync();

            await _context.Entry(nuevaConfig).Reference(c => c.Peluquero).LoadAsync();
            if (nuevaConfig.ServicioId.HasValue)
                await _context.Entry(nuevaConfig).Reference(c => c.Servicio).LoadAsync();

            // 2. Métricas de Turnos y Tasa de Cancelación
            int turnosCompletados = await _context.Turnos
                .CountAsync(t => t.FechaHoraInicio >= inicio && t.FechaHoraInicio <= fin && t.EstadoId == (int)EstadoTurnoEnum.Completado);

            int turnosCancelados = await _context.Turnos
                .CountAsync(t => t.FechaHoraInicio >= inicio && t.FechaHoraInicio <= fin && t.EstadoId == (int)EstadoTurnoEnum.Cancelado);

            int totalTurnosPeriodo = await _context.Turnos
                .CountAsync(t => t.FechaHoraInicio >= inicio && t.FechaHoraInicio <= fin);

            double tasaCancelacion = totalTurnosPeriodo > 0
                ? Math.Round((double)turnosCancelados / totalTurnosPeriodo * 100.0, 2)
                : 0.0;

            int turnosHoy = await _context.Turnos
                .CountAsync(t => t.FechaHoraInicio >= hoyInicio && t.FechaHoraInicio <= hoyFin
                              && (t.EstadoId == (int)EstadoTurnoEnum.Pendiente || t.EstadoId == (int)EstadoTurnoEnum.Confirmado));

            // 3. Servicio más vendido en el período
            string nombreServicioTop = "Sin datos";
            var topServicioGroup = await _context.DetallesVenta
                .Include(d => d.Venta)
                .Where(d => d.Venta != null && d.Venta.Fecha >= inicio && d.Venta.Fecha <= fin && d.TipoItem == TipoItemVentaEnum.Servicio)
                .GroupBy(d => d.ItemId)
                .Select(g => new { ServicioId = g.Key, Cantidad = g.Sum(x => x.Cantidad) })
                .OrderByDescending(x => x.Cantidad)
                .FirstOrDefaultAsync();

            if (topServicioGroup != null)
            {
                var serv = await _context.Servicios.AsNoTracking().FirstOrDefaultAsync(s => s.Id == topServicioGroup.ServicioId);
                if (serv != null)
                    nombreServicioTop = $"{serv.Nombre} ({topServicioGroup.Cantidad})";
            }

            // 4. Ranking de Peluqueros y Peluquero Estrella
            var rankingRaw = await _context.DetallesVenta
                .Include(d => d.Venta)
                .Where(d => d.Venta != null && d.Venta.Fecha >= inicio && d.Venta.Fecha <= fin && d.PeluqueroId.HasValue)
                .GroupBy(d => d.PeluqueroId!.Value)
                .Select(g => new
                {
                    PeluqueroId = g.Key,
                    ServiciosRealizados = g.Sum(x => x.Cantidad),
                    TotalFacturado = g.Sum(x => x.Subtotal)
                })
                .OrderByDescending(x => x.TotalFacturado)
                .ToListAsync();

            var peluquerosDict = await _context.Peluqueros
                .AsNoTracking()
                .ToDictionaryAsync(p => p.Id, p => $"{p.Nombre} {p.Apellido}".Trim());

            var rankingPeluqueros = rankingRaw.Select(r => new RankingPeluqueroDto
            {
                PeluqueroId = r.PeluqueroId,
                NombrePeluquero = peluquerosDict.TryGetValue(r.PeluqueroId, out var nom) ? nom : $"Peluquero #{r.PeluqueroId}",
                ServiciosRealizados = r.ServiciosRealizados,
                TotalFacturado = r.TotalFacturado
            }).ToList();

            string nombrePeluqueroTop = rankingPeluqueros.Any()
                ? $"{rankingPeluqueros.First().NombrePeluquero} (${rankingPeluqueros.First().TotalFacturado:N2})"
                : "Sin datos";

            // 5. Desglose de Ingresos por Medio de Pago
            var pagosRaw = await _context.Pagos
                .Include(p => p.Venta)
                .Include(p => p.MedioPago)
                .Where(p => p.Venta != null && p.Venta.Fecha >= inicio && p.Venta.Fecha <= fin)
                .GroupBy(p => p.MedioPago != null ? p.MedioPago.Nombre : "Otro")
                .Select(g => new IngresoPorMedioPagoDto
                {
                    MedioPago = g.Key,
                    TotalRecaudado = g.Sum(x => x.Importe),
                    CantidadOperaciones = g.Count()
                })
                .OrderByDescending(x => x.TotalRecaudado)
                .ToListAsync();

            // 6. Cantidad de productos en alerta de stock mínimo
            int stockCriticoCount = await _context.Productos
                .CountAsync(p => p.StockActual <= p.StockMinimoAlerta);

            var kpi = new DashboardKpiResponseDto
            {
                FechaDesde = inicio,
                FechaHasta = fin,
                FacturacionTotalMes = facturacion,
                CantidadVentasMes = cantidadVentas,
                TotalTurnosAtendidosMes = turnosCompletados,
                TurnosCanceladosPeriodo = turnosCancelados,
                TurnosPendientesHoy = turnosHoy,
                TasaCancelacionPorcentaje = tasaCancelacion,
                TicketPromedio = ticketPromedio,
                ServicioMasVendido = nombreServicioTop,
                PeluqueroEstrella = nombrePeluqueroTop,
                ProductosEnStockCritico = stockCriticoCount,
                IngresosPorMedioPago = pagosRaw,
                RankingPeluqueros = rankingPeluqueros
            };

            return ApiResponse<DashboardKpiResponseDto>.Exito(kpi, "KPIs gerenciales calculados exitosamente.");
        }
    }

                TurnoId = f.TurnoId,

        private static void SetOptionalProp(object target, string[] propNames, string? value)
        {
            var type = target.GetType();
            foreach (var name in propNames)
            {
                var prop = type.GetProperty(name);
                if (prop != null && prop.CanWrite && prop.PropertyType == typeof(string))
                {
                    prop.SetValue(target, value ?? string.Empty);
                    return;
                }
            }
        }

        private static string? GetOptionalProp(object target, string[] propNames)
        {
            var type = target.GetType();
            foreach (var name in propNames)
            {
                var prop = type.GetProperty(name);
                if (prop != null && prop.CanRead)
                {
                    return prop.GetValue(target)?.ToString();
                }
            }
            return null;
        }

        private static void SetOptionalDate(object target, string[] propNames, DateTime value)
        {
            var type = target.GetType();
            foreach (var name in propNames)
            {
                var prop = type.GetProperty(name);
                if (prop != null && prop.CanWrite && prop.PropertyType == typeof(DateTime))
                {
                    prop.SetValue(target, value);
                    return;
                }
            }
        }

        {
            var type = target.GetType();
            foreach (var name in propNames)
            {
                var prop = type.GetProperty(name);
                if (prop != null && prop.CanRead && prop.GetValue(target) is DateTime dt)
                {
                    return dt;
                }
            }
            return DateTime.UtcNow;
        }

        public async Task<ApiResponse<LiquidacionResponseDto>> LiquidarPeriodoAsync(GenerarLiquidacionDto dto)
        {
            var peluquero = await _context.Peluqueros
                .AsNoTracking()
                .FirstOrDefaultAsync(p => p.Id == dto.PeluqueroId);

            if (peluquero == null)
                return ApiResponse<LiquidacionResponseDto>.Falla("El profesional especificado no existe.");

            if (dto.FechaHasta < dto.FechaDesde)
                return ApiResponse<LiquidacionResponseDto>.Falla("La fecha de cierre del período no puede ser anterior a la fecha de inicio.");

            var configs = await _context.ComisionesConfigs
                .AsNoTracking()
                .Where(c => c.PeluqueroId == dto.PeluqueroId)
                .ToListAsync();

            var configGeneral = configs.FirstOrDefault(c => c.ServicioId == null);
            decimal pctServicioDefault = configGeneral?.PorcentajeServicio ?? 50.0m;
            decimal pctProductoDefault = configGeneral?.PorcentajeVentaProducto ?? 10.0m;

            var itemsPendientes = await _context.DetallesVenta
                .Include(d => d.Venta)
                .Where(d => d.PeluqueroId == dto.PeluqueroId
                            && d.Venta != null
                            && d.Venta.Fecha >= dto.FechaDesde
                            && d.Venta.Fecha <= dto.FechaHasta
                            && !_context.DetallesLiquidacion.Any(dl => dl.DetalleVentaId == d.Id))
                .ToListAsync();

            if (!itemsPendientes.Any())
            {
                return ApiResponse<LiquidacionResponseDto>.Falla(
                    "No existen servicios ni productos pendientes de liquidar para este profesional en el período seleccionado.");
            }

            decimal comisionAcumuladaServicios = 0;
            decimal comisionAcumuladaProductos = 0;

            var liquidacion = new Liquidacion
            {
                PeluqueroId = dto.PeluqueroId,
                FechaDesde = dto.FechaDesde,
                FechaHasta = dto.FechaHasta,
                FechaVencimiento = DateTime.UtcNow.AddDays(7),
                Estado = EstadoLiquidacionEnum.Pendiente
            };

            foreach (var item in itemsPendientes)
            {
                decimal montoComisionRenglon = 0;

                if (item.TipoItem == TipoItemVentaEnum.Servicio)
                {
                    var configEspecifica = configs.FirstOrDefault(c => c.ServicioId == item.ItemId);
                    decimal pctAplicar = configEspecifica?.PorcentajeServicio ?? pctServicioDefault;
                    montoComisionRenglon = Math.Round(item.Subtotal * (pctAplicar / 100m), 2);
                    comisionAcumuladaServicios += montoComisionRenglon;
                }
                else if (item.TipoItem == TipoItemVentaEnum.Producto)
                {
                    montoComisionRenglon = Math.Round(item.Subtotal * (pctProductoDefault / 100m), 2);
                    comisionAcumuladaProductos += montoComisionRenglon;
                }

                liquidacion.DetallesLiquidacion.Add(new DetalleLiquidacion
                {
                    DetalleVentaId = item.Id,
                    TurnoId = item.Venta?.TurnoId,
                    MontoComisionCalculado = montoComisionRenglon
                });
            }

            liquidacion.TotalServicios = comisionAcumuladaServicios;
            liquidacion.TotalProductos = comisionAcumuladaProductos;
            liquidacion.TotalPagar = comisionAcumuladaServicios + comisionAcumuladaProductos + dto.Bonificaciones - dto.Descuentos;

            await _context.Liquidaciones.AddAsync(liquidacion);
            await _unitOfWork.SaveChangesAsync();

            return ApiResponse<LiquidacionResponseDto>.Exito(new LiquidacionResponseDto
            {
                Id = liquidacion.Id,
                PeluqueroId = liquidacion.PeluqueroId,
                PeluqueroNombre = $"{peluquero.Nombre} {peluquero.Apellido}".Trim(),
                FechaDesde = liquidacion.FechaDesde,
                FechaHasta = liquidacion.FechaHasta,
                FechaVencimiento = liquidacion.FechaVencimiento,
                TotalServicios = liquidacion.TotalServicios,
                TotalProductos = liquidacion.TotalProductos,
                TotalPagar = liquidacion.TotalPagar,
                CantidadItemsLiquidados = itemsPendientes.Count,
                Estado = liquidacion.Estado.ToString()
            }, "Liquidación calculada y renglones bloqueados exitosamente.");
        }

        public async Task<ApiResponse<bool>> RegistrarPagoLiquidacionAsync(int liquidacionId, int? sesionCajaId = null)
        {
            var liq = await _context.Liquidaciones
                .Include(l => l.Peluquero)
                .FirstOrDefaultAsync(l => l.Id == liquidacionId);

            if (liq == null)
                return ApiResponse<bool>.Falla("La liquidación indicada no existe.");

            if (liq.Estado == EstadoLiquidacionEnum.Pagado)
                return ApiResponse<bool>.Falla("Esta liquidación ya fue abonada anteriormente.");

            liq.Estado = EstadoLiquidacionEnum.Pagado;
            liq.FechaPago = DateTime.UtcNow;

            if (sesionCajaId.HasValue && sesionCajaId.Value > 0)
            {
                var sesion = await _context.SesionesCaja.FirstOrDefaultAsync(s => s.Id == sesionCajaId.Value && s.EstadoAbierta);
                if (sesion != null)
                {
                    await _context.MovimientosCaja.AddAsync(new MovimientoCaja
                    {
                        SesionCajaId = sesion.Id,

                        Fecha = DateTime.UtcNow
                    });
                }
            }

            _context.Liquidaciones.Update(liq);
            await _unitOfWork.SaveChangesAsync();

            return ApiResponse<bool>.Exito(true, "Pago de liquidación asentado correctamente.");

    {
        private readonly IOrdenCompraRepository _ordenCompraRepo;
        private readonly IUnitOfWork _uow;
        private readonly ApplicationDbContext _context;

        public OrdenCompraService(
            IOrdenCompraRepository ordenCompraRepo,
            IUnitOfWork uow,
            ApplicationDbContext context)
        {
            _ordenCompraRepo = ordenCompraRepo;
            _uow = uow;
            _context = context;
        }

        public async Task<ApiResponse<IReadOnlyList<OrdenCompraResponseDto>>> ObtenerTodasAsync(
            int? proveedorId = null,
            EstadoOrdenCompraEnum? estado = null,
            CancellationToken ct = default)
        {
            var ordenes = await _ordenCompraRepo.GetAllAsync(proveedorId, estado, ct);
            return ApiResponse<IReadOnlyList<OrdenCompraResponseDto>>.Exito(ordenes.Select(MapToDto).ToList());
        }

        public async Task<ApiResponse<OrdenCompraResponseDto>> ObtenerPorIdAsync(int id, CancellationToken ct = default)
        {
            var orden = await _ordenCompraRepo.GetByIdCompletaAsync(id, ct);
            if (orden == null)
                return ApiResponse<OrdenCompraResponseDto>.Falla($"La orden de compra #{id} no existe.");

            return ApiResponse<OrdenCompraResponseDto>.Exito(MapToDto(orden));
        }

        public async Task<ApiResponse<OrdenCompraResponseDto>> CrearOrdenAsync(
            CrearOrdenCompraDto dto,
            int usuarioId,
            CancellationToken ct = default)
        {
            if (dto.Detalles == null || !dto.Detalles.Any())
                return ApiResponse<OrdenCompraResponseDto>.Falla("La orden de compra debe tener al menos un ítem.");

            var orden = new OrdenCompra
            {
                ProveedorId = dto.ProveedorId,
                NumeroComprobanteProveedor = dto.NumeroComprobanteProveedor,
                Observaciones = dto.Observaciones,
                FechaEmision = DateTime.UtcNow,
                Estado = EstadoOrdenCompraEnum.Emitida,
                UsuarioCreacionId = usuarioId,
                CreatedByUserId = usuarioId
            };

            decimal total = 0;
            foreach (var item in dto.Detalles)
            {
                var producto = await _uow.Productos.GetByIdAsync(item.ProductoId);
                if (producto == null)
                    return ApiResponse<OrdenCompraResponseDto>.Falla($"El producto con ID {item.ProductoId} no existe.");

                decimal subtotal = item.CantidadPedida * item.PrecioUnitarioCosto;
                total += subtotal;

                orden.Detalles.Add(new DetalleOrdenCompra
                {
                    ProductoId = item.ProductoId,
                    CantidadPedida = item.CantidadPedida,
                    CantidadRecibida = 0,
                    PrecioUnitarioCosto = item.PrecioUnitarioCosto,
                    Subtotal = subtotal,
                    CreatedByUserId = usuarioId
                });
            }

            orden.Total = total;

            await _ordenCompraRepo.AddAsync(orden, ct);
            await _uow.SaveChangesAsync(ct);

            var creada = await _ordenCompraRepo.GetByIdCompletaAsync(orden.Id, ct);
            return ApiResponse<OrdenCompraResponseDto>.Exito(MapToDto(creada ?? orden), "Orden de compra emitida correctamente.");
        }

        public async Task<ApiResponse<OrdenCompraResponseDto>> RecepcionarOrdenAsync(
            int id,
            RecepcionarOrdenCompraDto dto,
            int usuarioId,
            CancellationToken ct = default)
        {
            var orden = await _ordenCompraRepo.GetByIdCompletaAsync(id, ct);
            if (orden == null)
                return ApiResponse<OrdenCompraResponseDto>.Falla("La orden de compra no existe.");

            if (orden.Estado != EstadoOrdenCompraEnum.Emitida && orden.Estado != EstadoOrdenCompraEnum.Borrador)
                return ApiResponse<OrdenCompraResponseDto>.Falla($"No se puede recepcionar una orden en estado {orden.Estado}.");

            await _uow.BeginTransactionAsync(ct);
            try
            {
                decimal totalReal = 0;

                foreach (var detalle in orden.Detalles)
                {
                    var overrideItem = dto.CantidadesRecibidas?.FirstOrDefault(x => x.ProductoId == detalle.ProductoId);
                    int cantRecibida = overrideItem?.CantidadRecibida ?? detalle.CantidadPedida;

                    detalle.CantidadRecibida = cantRecibida;
                    detalle.Subtotal = cantRecibida * detalle.PrecioUnitarioCosto;
                    totalReal += detalle.Subtotal;

                    if (cantRecibida > 0)
                    {
                        var producto = await _uow.Productos.GetByIdAsync(detalle.ProductoId);
                        if (producto != null)
                        {
                            // Trazabilidad histórica de precios si cambió el costo
                            if (dto.ActualizarCostoProductos && producto.PrecioCosto != detalle.PrecioUnitarioCosto)
                            {
                                decimal nuevoPrecioVenta = overrideItem?.NuevoPrecioVentaSugerido ?? producto.PrecioVenta;

                                await _context.HistorialPreciosProducto.AddAsync(new HistorialPrecioProducto
                                {
                                    ProductoId = producto.Id,
                                    PrecioCostoAnterior = producto.PrecioCosto,
                                    PrecioCostoNuevo = detalle.PrecioUnitarioCosto,
                                    PrecioVentaAnterior = producto.PrecioVenta,
                                    PrecioVentaNuevo = nuevoPrecioVenta,
                                    FechaCambio = DateTime.UtcNow,
                                    UsuarioModificacionId = usuarioId,
                                    Motivo = $"Recepción Orden de Compra #{orden.Id}"
                                }, ct);

                                producto.PrecioCosto = detalle.PrecioUnitarioCosto;
                                producto.PrecioVenta = nuevoPrecioVenta;
                            }

                            producto.StockActual += cantRecibida;
                            await _uow.Productos.UpdateAsync(producto);

                            // Registro en Kardex (MovimientoStock)
                            await _uow.Productos.RegistrarMovimientoAsync(new MovimientoStock
                            {
                                ProductoId = producto.Id,
                                TipoMovimiento = TipoMovimientoStockEnum.IngresoCompra,
                                Cantidad = cantRecibida,
                                Fecha = DateTime.UtcNow,
                                UsuarioId = usuarioId,
                                Motivo = $"Ingreso por Orden de Compra #{orden.Id} ({dto.NumeroComprobanteProveedor ?? orden.NumeroComprobanteProveedor ?? "S/N"})"
                            });
                        }
                    }
                }

                orden.Total = totalReal;
                orden.Estado = EstadoOrdenCompraEnum.Recibida;
                orden.FechaRecepcion = DateTime.UtcNow;
                orden.UsuarioRecepcionId = usuarioId;
                if (!string.IsNullOrWhiteSpace(dto.NumeroComprobanteProveedor))
                    orden.NumeroComprobanteProveedor = dto.NumeroComprobanteProveedor;

                // Impacto opcional en Caja Abierta
                if (dto.ImputarEgresoEnCajaAbierta && totalReal > 0)
                {
                    var cajaAbierta = await _uow.SesionesCaja.GetCajaAbiertaActualAsync(ct);
                    if (cajaAbierta == null)
                    {
                        await _uow.RollbackTransactionAsync(ct);
                        return ApiResponse<OrdenCompraResponseDto>.Falla("No hay una sesión de caja abierta para imputar el egreso del pago al proveedor.");
                    }

                    await _uow.SesionesCaja.RegistrarMovimientoCajaAsync(new MovimientoCaja
                    {
                        SesionCajaId = cajaAbierta.Id,
                        Tipo = "PagoProveedor",
                        Monto = totalReal,
                        Concepto = $"Pago Orden de Compra #{orden.Id} - Prov: {orden.Proveedor?.RazonSocial}",
                        Fecha = DateTime.UtcNow,
                        CreatedByUserId = usuarioId
                    }, ct);
                }

                _ordenCompraRepo.Update(orden);
                await _uow.CommitTransactionAsync(ct);

                return ApiResponse<OrdenCompraResponseDto>.Exito(MapToDto(orden), "Mercadería recepcionada y stock actualizado exitosamente.");
            }
            catch (Exception ex)
            {
                await _uow.RollbackTransactionAsync(ct);
                return ApiResponse<OrdenCompraResponseDto>.Falla($"Error al recepcionar la orden de compra: {ex.Message}");
            }
        }

        public async Task<ApiResponse<bool>> CancelarOrdenAsync(int id, string motivo, int usuarioId, CancellationToken ct = default)
        {
            var orden = await _ordenCompraRepo.GetByIdCompletaAsync(id, ct);
            if (orden == null)
                return ApiResponse<bool>.Falla("La orden de compra no existe.");

            if (orden.Estado == EstadoOrdenCompraEnum.Recibida)
                return ApiResponse<bool>.Falla("No se puede cancelar una orden que ya fue recepcionada en stock.");

            orden.Estado = EstadoOrdenCompraEnum.Cancelada;
            orden.Observaciones = $"{orden.Observaciones} | Cancelada: {motivo}".Trim();
            _ordenCompraRepo.Update(orden);
            await _uow.SaveChangesAsync(ct);

            return ApiResponse<bool>.Exito(true, "Orden de compra cancelada correctamente.");
        }

        private static OrdenCompraResponseDto MapToDto(OrdenCompra o)
        {
            return new OrdenCompraResponseDto
            {
                Id = o.Id,
                ProveedorId = o.ProveedorId,
                ProveedorRazonSocial = o.Proveedor?.RazonSocial ?? string.Empty,
                NumeroComprobanteProveedor = o.NumeroComprobanteProveedor,
                FechaEmision = o.FechaEmision,
                FechaRecepcion = o.FechaRecepcion,
                Estado = o.Estado,
                EstadoNombre = o.Estado.ToString(),
                Total = o.Total,
                Observaciones = o.Observaciones,
                Detalles = o.Detalles.Select(d => new DetalleOrdenCompraResponseDto
                {
                    Id = d.Id,
                    ProductoId = d.ProductoId,
                    ProductoNombre = d.Producto?.Nombre ?? string.Empty,
                    CantidadPedida = d.CantidadPedida,
                    CantidadRecibida = d.CantidadRecibida,
                    PrecioUnitarioCosto = d.PrecioUnitarioCosto,
                    Subtotal = d.Subtotal
                }).ToList()
            };
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using AutoMapper;
using Microsoft.EntityFrameworkCore;
using MiPeluqueria.Api.Data.Context;
using MiPeluqueria.Api.DTOs.Common;
using MiPeluqueria.Api.DTOs.Turnos;
using MiPeluqueria.Api.Models.Turnos;
using MiPeluqueria.Api.Repositories.Interfaces;
using MiPeluqueria.Api.Services.Interfaces;

namespace MiPeluqueria.Api.Services.Implementations
{
    public class PeluqueroService : IPeluqueroService
    {
        private readonly IUnitOfWork _unitOfWork;
        private readonly ApplicationDbContext _context;
        private readonly IMapper _mapper;

        public PeluqueroService(IUnitOfWork unitOfWork, ApplicationDbContext context, IMapper mapper)
        {
            _unitOfWork = unitOfWork;
            _context = context;
            _mapper = mapper;
        }

        public async Task<ApiResponse<IEnumerable<PeluqueroDto>>> GetAllAsync()
        {
            var peluqueros = await _context.Peluqueros
                .Include(p => p.PeluqueroEspecialidades)
                .ThenInclude(pe => pe.Especialidad)
                .AsNoTracking()
                .ToListAsync();

            var dtos = _mapper.Map<IEnumerable<PeluqueroDto>>(peluqueros);
            return ApiResponse<IEnumerable<PeluqueroDto>>.Exito(dtos);
        }

        public async Task<ApiResponse<PeluqueroDto>> GetByIdAsync(int id)
        {
            var peluquero = await _context.Peluqueros
                .Include(p => p.PeluqueroEspecialidades)
                .ThenInclude(pe => pe.Especialidad)
                .FirstOrDefaultAsync(p => p.Id == id);

            if (peluquero == null)
                return ApiResponse<PeluqueroDto>.Falla("El peluquero solicitado no existe.");

            return ApiResponse<PeluqueroDto>.Exito(_mapper.Map<PeluqueroDto>(peluquero));
        }

        public async Task<ApiResponse<PeluqueroDto>> CreateAsync(CrearPeluqueroDto dto)
        {
            var peluquero = _mapper.Map<Peluquero>(dto);
            await _context.Peluqueros.AddAsync(peluquero);
            await _unitOfWork.SaveChangesAsync();

            var especialidadesIds = GetEspecialidadesIdsFromDto(dto);
            if (especialidadesIds.Any())
            {
                foreach (var espId in especialidadesIds.Distinct())
                {
                    await _context.PeluqueroEspecialidades.AddAsync(new PeluqueroEspecialidad
                    {
                        PeluqueroId = peluquero.Id,
                        EspecialidadId = espId
                    });
                }
                await _unitOfWork.SaveChangesAsync();
            }

            return await GetByIdAsync(peluquero.Id);
        }

        public async Task<ApiResponse<PeluqueroDto>> UpdateAsync(int id, CrearPeluqueroDto dto)
        {
            var peluquero = await _context.Peluqueros
                .Include(p => p.PeluqueroEspecialidades)
                .FirstOrDefaultAsync(p => p.Id == id);

            if (peluquero == null)
                return ApiResponse<PeluqueroDto>.Falla("El peluquero solicitado no existe.");

            _mapper.Map(dto, peluquero);

            var especialidadesIds = GetEspecialidadesIdsFromDto(dto);
            if (especialidadesIds.Any())
            {
                _context.PeluqueroEspecialidades.RemoveRange(peluquero.PeluqueroEspecialidades);
                foreach (var espId in especialidadesIds.Distinct())
                {
                    await _context.PeluqueroEspecialidades.AddAsync(new PeluqueroEspecialidad
                    {
                        PeluqueroId = peluquero.Id,
                        EspecialidadId = espId
                    });
                }
            }

            _context.Peluqueros.Update(peluquero);
            await _unitOfWork.SaveChangesAsync();

            return await GetByIdAsync(id);
        }

        public async Task<ApiResponse<bool>> DeleteAsync(int id)
        {
            var peluquero = await _context.Peluqueros.FirstOrDefaultAsync(p => p.Id == id);
            if (peluquero == null)
                return ApiResponse<bool>.Falla("El peluquero solicitado no existe.");

            peluquero.IsDeleted = true;
            SetOptionalProp(peluquero, "Activo", false);

            _context.Peluqueros.Update(peluquero);
            await _unitOfWork.SaveChangesAsync();

            return ApiResponse<bool>.Exito(true, "Peluquero dado de baja correctamente.");
        }

        public async Task<ApiResponse<IEnumerable<HorarioLaboralDto>>> ObtenerHorariosAsync(int peluqueroId)
        {
            var horarios = await _context.HorariosLaborales
                .Where(h => h.PeluqueroId == peluqueroId)
                .AsNoTracking()
                .ToListAsync();

            var dtos = horarios.Select(h => new HorarioLaboralDto
            {
                Id = h.Id,
                PeluqueroId = h.PeluqueroId,
                DiaSemana = Convert.ToInt32(GetOptionalProp(h, new[] { "DiaSemana", "Dia" }) ?? 1),
                HoraInicio = GetTimeSpanProp(h, new[] { "HoraInicio", "HoraDesde", "HoraApertura" }) ?? new TimeSpan(9, 0, 0),
                HoraFin = GetTimeSpanProp(h, new[] { "HoraFin", "HoraHasta", "HoraCierre" }) ?? new TimeSpan(18, 0, 0),
                HoraInicioDescanso = GetTimeSpanProp(h, new[] { "HoraInicioDescanso", "DescansoInicio" }),
                HoraFinDescanso = GetTimeSpanProp(h, new[] { "HoraFinDescanso", "DescansoFin" }),
                Activo = (bool?)(GetOptionalProp(h, new[] { "Activo", "EsLaborable" })) ?? true
            })
            .OrderBy(h => h.DiaSemana)
            .ToList();

            return ApiResponse<IEnumerable<HorarioLaboralDto>>.Exito(dtos);
        }

        public async Task<ApiResponse<bool>> ConfigurarHorariosAsync(int peluqueroId, GuardarHorariosPeluqueroDto dto)
        {
            var peluquero = await _context.Peluqueros.FirstOrDefaultAsync(p => p.Id == peluqueroId);
            if (peluquero == null)
                return ApiResponse<bool>.Falla("El peluquero especificado no existe.");

            var existentes = await _context.HorariosLaborales
                .Where(h => h.PeluqueroId == peluqueroId)
                .ToListAsync();

            if (existentes.Any())
                _context.HorariosLaborales.RemoveRange(existentes);

            foreach (var item in dto.Horarios)
            {
                var nuevo = new HorarioLaboral
                {
                    PeluqueroId = peluqueroId
                };

                SetDiaProp(nuevo, new[] { "DiaSemana", "Dia" }, item.DiaSemana);
                SetOptionalProp(nuevo, "HoraInicio", item.HoraInicio);
                SetOptionalProp(nuevo, "HoraDesde", item.HoraInicio);
                SetOptionalProp(nuevo, "HoraFin", item.HoraFin);
                SetOptionalProp(nuevo, "HoraHasta", item.HoraFin);
                if (item.HoraInicioDescanso.HasValue)
                    SetOptionalProp(nuevo, "HoraInicioDescanso", item.HoraInicioDescanso.Value);
                if (item.HoraFinDescanso.HasValue)
                    SetOptionalProp(nuevo, "HoraFinDescanso", item.HoraFinDescanso.Value);
                SetOptionalProp(nuevo, "Activo", item.Activo);

                await _context.HorariosLaborales.AddAsync(nuevo);
            }

            await _unitOfWork.SaveChangesAsync();
            return ApiResponse<bool>.Exito(true, "Horarios laborales actualizados correctamente.");
        }

        private static List<int> GetEspecialidadesIdsFromDto(CrearPeluqueroDto dto)
        {
            var prop = dto.GetType().GetProperty("EspecialidadesIds") ?? dto.GetType().GetProperty("Especialidades");
            if (prop != null && prop.GetValue(dto) is IEnumerable<int> ids)
                return ids.ToList();
            return new List<int>();
        }

        private static object? GetOptionalProp(object target, string[] names)
        {
            var type = target.GetType();
            foreach (var name in names)
            {
                var prop = type.GetProperty(name);
                if (prop != null && prop.CanRead)
                    return prop.GetValue(target);
            }
            return null;
        }

        private static TimeSpan? GetTimeSpanProp(object target, string[] names)
        {
            var val = GetOptionalProp(target, names);
            if (val is TimeSpan ts) return ts;
            if (val is TimeOnly to) return to.ToTimeSpan();
            return null;
        }

        private static void SetOptionalProp(object target, string name, object value)
        {
            var prop = target.GetType().GetProperty(name);
            if (prop != null && prop.CanWrite)
            {
                try { prop.SetValue(target, value); } catch { }
            }
        }

        private static void SetDiaProp(object target, string[] names, int dia)
        {
            var type = target.GetType();
            foreach (var name in names)
            {
                var prop = type.GetProperty(name);
                if (prop != null && prop.CanWrite)
                {
                    if (prop.PropertyType.IsEnum)
                        prop.SetValue(target, Enum.ToObject(prop.PropertyType, dia));
                    else
                        prop.SetValue(target, Convert.ChangeType(dia, prop.PropertyType));
                    return;
                }
            }
        }
    }

using AutoMapper;
using Microsoft.EntityFrameworkCore;
using MiPeluqueria.Api.Data.Context;

        private readonly IMapper _mapper;

            _mapper = mapper;
        }

        public async Task<ApiResponse<IEnumerable<ProductoDto>>> GetAllAsync(bool soloBajoStock = false)
        {
            if (soloBajoStock)
                return await GetStockCriticoAsync();

            var productos = await _context.Productos
                .AsNoTracking()
                .ToListAsync();

            var dtos = _mapper.Map<IEnumerable<ProductoDto>>(productos);
            return ApiResponse<IEnumerable<ProductoDto>>.Ok(dtos);
        }

        public async Task<ApiResponse<IEnumerable<ProductoDto>>> GetStockCriticoAsync()
        {
            var productos = await _context.Productos
                .Where(p => p.StockActual <= p.StockMinimoAlerta)
                .AsNoTracking()
                .ToListAsync();

            var dtos = _mapper.Map<IEnumerable<ProductoDto>>(productos);
            return ApiResponse<IEnumerable<ProductoDto>>.Ok(dtos);
        }

        public async Task<ApiResponse<ProductoDto>> GetByIdAsync(int id)
        {
            var producto = await _context.Productos
                .FirstOrDefaultAsync(p => p.Id == id);

            if (producto == null)
                return ApiResponse<ProductoDto>.Fail("El producto solicitado no existe.");

            return ApiResponse<ProductoDto>.Ok(_mapper.Map<ProductoDto>(producto));
        }

        public async Task<ApiResponse<ProductoDto>> CreateAsync(CrearProductoDto dto, int usuarioId = 1)
        {
            var producto = _mapper.Map<Producto>(dto);
            await _context.Productos.AddAsync(producto);
            await _context.SaveChangesAsync();

            if (producto.StockActual > 0)
            {
                var movInicial = new MovimientoStock
                {
                    ProductoId = producto.Id,
                    TipoMovimiento = (TipoMovimientoStockEnum)1,
                    Cantidad = producto.StockActual,
                    Motivo = "Inventario inicial de alta de producto"
                };
                SetOptionalProp(movInicial, "UsuarioId", usuarioId);
                SetOptionalProp(movInicial, "StockResultante", producto.StockActual);

                await _context.MovimientosStock.AddAsync(movInicial);
                await _context.SaveChangesAsync();
            }

            return ApiResponse<ProductoDto>.Ok(_mapper.Map<ProductoDto>(producto), "Producto creado exitosamente.");
        }

        public async Task<ApiResponse<ProductoDto>> UpdateAsync(int id, CrearProductoDto dto)
        {
            var producto = await _context.Productos.FirstOrDefaultAsync(p => p.Id == id);
            if (producto == null)
                return ApiResponse<ProductoDto>.Fail("El producto solicitado no existe.");

            var stockActualPrevio = producto.StockActual;
            _mapper.Map(dto, producto);
            producto.StockActual = stockActualPrevio;

            _context.Productos.Update(producto);
            await _context.SaveChangesAsync();

            return ApiResponse<ProductoDto>.Ok(_mapper.Map<ProductoDto>(producto), "Producto actualizado correctamente.");
        }

        public async Task<ApiResponse<bool>> DeleteAsync(int id)
        {
            var producto = await _context.Productos.FirstOrDefaultAsync(p => p.Id == id);
            if (producto == null)
                return ApiResponse<bool>.Fail("El producto solicitado no existe.");

            producto.IsDeleted = true;
            await _context.SaveChangesAsync();

            return ApiResponse<bool>.Ok(true, "Producto dado de baja correctamente.");
        }

        public async Task<ApiResponse<bool>> RegistrarMovimientoStockAsync(MovimientoStockDto dto, int usuarioId = 1)
        {
            var producto = await _context.Productos.FirstOrDefaultAsync(p => p.Id == dto.ProductoId);
            if (producto == null)
                return ApiResponse<bool>.Fail("El producto especificado no existe.");

            var nombreTipo = dto.TipoMovimiento.ToString().ToLower();
            bool esEgreso = nombreTipo.Contains("egreso") ||
                            nombreTipo.Contains("salida") ||
                            nombreTipo.Contains("venta") ||
                            nombreTipo.Contains("consumo") ||
                            nombreTipo.Contains("negativo") ||
                            Convert.ToInt32(dto.TipoMovimiento) == 2;

            int delta = esEgreso ? -Math.Abs(dto.Cantidad) : Math.Abs(dto.Cantidad);

            if (producto.StockActual + delta < 0)
                return ApiResponse<bool>.Fail($"Stock insuficiente. Stock actual: {producto.StockActual}.");

            producto.StockActual += delta;
            _context.Productos.Update(producto);

            var movimiento = new MovimientoStock
            {
                ProductoId = producto.Id,
                TipoMovimiento = dto.TipoMovimiento,
                Cantidad = Math.Abs(dto.Cantidad),
                Motivo = dto.Motivo
            };
            SetOptionalProp(movimiento, "UsuarioId", usuarioId);
            SetOptionalProp(movimiento, "StockResultante", producto.StockActual);

            await _context.MovimientosStock.AddAsync(movimiento);
            await _context.SaveChangesAsync();

            return ApiResponse<bool>.Ok(true, $"Stock actualizado correctamente. Stock resultante: {producto.StockActual}.");
        }

                return ApiResponse<IEnumerable<object>>.Fail("El producto solicitado no existe.");

            var movimientos = await _context.MovimientosStock
                .Where(m => m.ProductoId == productoId)
                .OrderByDescending(m => m.Id)
                .AsNoTracking()
                .ToListAsync();

            var resultado = movimientos.Select(m => (object)new
            {
                m.Id,
                m.ProductoId,
                TipoMovimiento = m.TipoMovimiento.ToString(),
                m.Cantidad,
                m.Motivo
            }).ToList();

            return ApiResponse<IEnumerable<object>>.Ok(resultado);
        }

        {
            var prop = target.GetType().GetProperty(propName);
            if (prop != null && prop.CanWrite)
            {
                prop.SetValue(target, value);
            }

        public async Task<ApiResponse<bool>> GuardarRecetaServicioAsync(ConfigurarRecetaServicioDto dto, int usuarioId, CancellationToken ct = default)
        {
            var existentes = await _context.Set<ServicioInsumoReceta>()
                .Where(r => r.ServicioId == dto.ServicioId)
                .ToListAsync(ct);

            foreach (var ex in existentes)
            {
                ex.IsDeleted = true;
                ex.DeletedAt = DateTime.UtcNow;
            }

            foreach (var item in dto.Insumos)
            {
                await _context.Set<ServicioInsumoReceta>().AddAsync(new ServicioInsumoReceta
                {
                    ServicioId = dto.ServicioId,
                    ProductoId = item.ProductoId,
                    CantidadUnidadesDescontar = item.CantidadUnidadesDescontar,
                    CantidadMedidaEstandar = item.CantidadMedidaEstandar,
                    UnidadMedida = item.UnidadMedida,
                    DescuentoAutomaticoAlCompletarTurno = item.DescuentoAutomaticoAlCompletarTurno,
                    CreatedByUserId = usuarioId
                }, ct);

            await _uow.SaveChangesAsync(ct);
            return ApiResponse<bool>.Exito(true, "Receta técnica de insumos actualizada correctamente.");

        public async Task<ApiResponse<int>> AplicarConsumoGabinetePorTurnoAsync(int turnoId, int usuarioId, CancellationToken ct = default)
        {
            var turno = await _uow.Turnos.GetByIdConDetallesAsync(turnoId);
            if (turno == null)
                return ApiResponse<int>.Falla("El turno indicado no existe.");

            var serviciosIds = turno.TurnoServicios.Select(ts => ts.ServicioId).Distinct().ToList();
            if (!serviciosIds.Any())
                return ApiResponse<int>.Exito(0, "El turno no posee servicios asociados.");

            var recetas = await _context.Set<ServicioInsumoReceta>()
                .Where(r => serviciosIds.Contains(r.ServicioId) && r.DescuentoAutomaticoAlCompletarTurno)
                .ToListAsync(ct);

            if (!recetas.Any())
                return ApiResponse<int>.Exito(0, "Los servicios del turno no requieren descuento automático de insumos.");

            int movimientosGenerados = 0;
            foreach (var receta in recetas)
            {
                if (receta.CantidadUnidadesDescontar <= 0) continue;

                var producto = await _uow.Productos.GetByIdAsync(receta.ProductoId);
                if (producto != null)
                {
                    producto.StockActual = Math.Max(0, producto.StockActual - receta.CantidadUnidadesDescontar);
                    await _uow.Productos.UpdateAsync(producto);

                    await _uow.Productos.RegistrarMovimientoAsync(new MovimientoStock
                    {
                        ProductoId = producto.Id,
                        TipoMovimiento = TipoMovimientoStockEnum.ConsumoInternoSalon,
                        Cantidad = receta.CantidadUnidadesDescontar,
                        Fecha = DateTime.UtcNow,
                        UsuarioId = usuarioId,
                        Motivo = $"Consumo técnico automático por Turno #{turno.Id} ({receta.CantidadMedidaEstandar} {receta.UnidadMedida})"
                    });
                    movimientosGenerados++;
                }
            }

            await _uow.SaveChangesAsync(ct);
            return ApiResponse<int>.Exito(movimientosGenerados, $"Se descontaron {movimientosGenerados} insumos de gabinete.");
        }
    }

        public async Task<ApiResponse<TurnoResponseDto>> CreateAsync(CrearTurnoDto dto, int usuarioId = 1)
        {
            var cliente = await _context.Clientes.FindAsync(dto.ClienteId);
            if (cliente == null)
                return ApiResponse<TurnoResponseDto>.Falla("El cliente especificado no existe.");

            var peluquero = await _context.Peluqueros.FindAsync(dto.PeluqueroId);
            if (peluquero == null)
                return ApiResponse<TurnoResponseDto>.Falla("El peluquero especificado no existe.");

            var serviciosIds = dto.ServiciosIds ?? new List<int>();
            int duracionTotal = 30;
            decimal montoTotalEstimado = 0m;
            var servicios = new List<Servicio>();

            if (serviciosIds.Any())
            {
                servicios = await _context.Servicios
                    .Where(s => serviciosIds.Contains(s.Id) && s.Activo)
                    .ToListAsync();

                if (!servicios.Any())
                    return ApiResponse<TurnoResponseDto>.Falla("Los servicios seleccionados no existen o están inactivos.");

                duracionTotal = servicios.Sum(s => s.DuracionMinutos > 0 ? s.DuracionMinutos : 30);
                montoTotalEstimado = servicios.Sum(s => s.Precio);
            }

            var fechaInicio = dto.FechaHoraInicio;
            var fechaFin = fechaInicio.AddMinutes(duracionTotal);

            bool existeSolapamiento = await _context.Turnos.AnyAsync(t =>
                t.PeluqueroId == dto.PeluqueroId &&
                t.EstadoId != (int)EstadoTurnoEnum.Cancelado &&
                fechaInicio < t.FechaHoraFin &&
                fechaFin > t.FechaHoraInicio);

            if (existeSolapamiento)
                return ApiResponse<TurnoResponseDto>.Falla("El peluquero ya posee otro turno asignado en ese rango horario.");

            var turno = new Turno
            {
                ClienteId = dto.ClienteId,
                PeluqueroId = dto.PeluqueroId,
                FechaHoraInicio = fechaInicio,
                FechaHoraFin = fechaFin,
                EstadoId = (int)EstadoTurnoEnum.Pendiente,
                MontoTotalEstimado = montoTotalEstimado,
                Observaciones = dto.Observaciones,
                CreatedByUserId = usuarioId
            };

            await _context.Turnos.AddAsync(turno);
            await _unitOfWork.SaveChangesAsync();

            foreach (var serv in servicios)
            {
                var ts = new TurnoServicio
                {
                    TurnoId = turno.Id,
                    ServicioId = serv.Id
                };
                SetOptionalProp(ts, "PrecioAcordado", serv.Precio);
                SetOptionalProp(ts, "Precio", serv.Precio);

                EstadoAnteriorId = (int)EstadoTurnoEnum.Pendiente,
                EstadoNuevoId = (int)EstadoTurnoEnum.Pendiente,

                UsuarioId = usuarioId,
                Observaciones = "Alta inicial de reserva de turno."
            });

            await _unitOfWork.SaveChangesAsync();
            return await GetByIdAsync(turno.Id);

        public async Task<ApiResponse<bool>> CambiarEstadoAsync(int id, CambiarEstadoTurnoDto dto, int usuarioId = 1)

            int nuevoEstado = dto.NuevoEstadoId > 0 ? dto.NuevoEstadoId : (int)EstadoTurnoEnum.Confirmado;

            turno.EstadoId = nuevoEstado;

            if (estadoAnterior != (int)EstadoTurnoEnum.Completado && nuevoEstado == (int)EstadoTurnoEnum.Completado)

        public async Task<ApiResponse<IEnumerable<string>>> ObtenerHorariosDisponiblesAsync(int peluqueroId, DateTime fecha, List<int>? serviciosIds = null)
        {
            var peluquero = await _context.Peluqueros.FindAsync(peluqueroId);
            if (peluquero == null)
                return ApiResponse<IEnumerable<string>>.Falla("El peluquero especificado no existe.");

            int duracionTotalMinutos = 30;
            if (serviciosIds != null && serviciosIds.Any())
            {
                var servicios = await _context.Servicios
                    .Where(s => serviciosIds.Contains(s.Id))
                    .AsNoTracking()
                    .ToListAsync();

                if (servicios.Any())
                    duracionTotalMinutos = servicios.Sum(s => s.DuracionMinutos > 0 ? s.DuracionMinutos : 30);
            }

            int diaSemana = (int)fecha.DayOfWeek;
            var horariosPeluquero = await _context.HorariosLaborales
                .Where(h => h.PeluqueroId == peluqueroId)
                .AsNoTracking()
                .ToListAsync();

            var horarioDia = horariosPeluquero.FirstOrDefault(h => (int)h.DiaSemana == diaSemana);

            TimeSpan horaApertura = new TimeSpan(9, 0, 0);
            TimeSpan horaCierre = new TimeSpan(20, 0, 0);
            TimeSpan? descansoInicio = null;
            TimeSpan? descansoFin = null;

            if (horarioDia != null)

                if (!activo)

                horaApertura = horarioDia.HoraInicio;
                horaCierre = horarioDia.HoraFin;
                descansoInicio = GetTimeSpanProp(horarioDia, new[] { "HoraInicioDescanso", "DescansoInicio" });
                descansoFin = GetTimeSpanProp(horarioDia, new[] { "HoraFinDescanso", "DescansoFin" });

            var finDia = inicioDia.AddDays(1).AddTicks(-1);

            var turnosOcupados = await _context.Turnos
                .Where(t => t.PeluqueroId == peluqueroId
                         && t.FechaHoraInicio >= inicioDia
                         && t.FechaHoraInicio <= finDia
                         && t.EstadoId != (int)EstadoTurnoEnum.Cancelado)
                .Select(t => new { t.FechaHoraInicio, t.FechaHoraFin })
                .AsNoTracking()
                .ToListAsync();

            var bloqueosPeluquero = await _context.BloqueosHorario
                .Where(b => b.PeluqueroId == peluqueroId)
                .AsNoTracking()
                .ToListAsync();

            var slotsDisponibles = new List<string>();
            var cursor = fecha.Date.Add(horaApertura);
            var limiteJornada = fecha.Date.Add(horaCierre);

            while (cursor.AddMinutes(duracionTotalMinutos) <= limiteJornada)
            {
                var slotInicio = cursor;
                var slotFin = cursor.AddMinutes(duracionTotalMinutos);

                bool enDescanso = false;
                if (descansoInicio.HasValue && descansoFin.HasValue)
                {
                    var dInicio = fecha.Date.Add(descansoInicio.Value);
                    var dFin = fecha.Date.Add(descansoFin.Value);
                    if (slotInicio < dFin && slotFin > dInicio)
                        enDescanso = true;
                }

                bool solapaTurno = turnosOcupados.Any(t => slotInicio < t.FechaHoraFin && slotFin > t.FechaHoraInicio);
                bool solapaBloqueo = bloqueosPeluquero.Any(b => slotInicio < b.FechaFin && slotFin > b.FechaInicio);

                if (!enDescanso && !solapaTurno && !solapaBloqueo)
                {
                    slotsDisponibles.Add(slotInicio.ToString("HH:mm"));
                }

                cursor = cursor.AddMinutes(30);
            }

            return ApiResponse<IEnumerable<string>>.Exito(slotsDisponibles);
        }

        public async Task<ApiResponse<IEnumerable<TurnoResponseDto>>> ObtenerTurnosPorClienteAsync(int clienteId)
        {
            var turnos = await _context.Turnos
                .Include(t => t.Cliente)
                .Include(t => t.Peluquero)
                .Include(t => t.Estado)
                .Include(t => t.TurnoServicios)
                    .ThenInclude(ts => ts.Servicio)
                .Where(t => t.ClienteId == clienteId)
                .OrderByDescending(t => t.FechaHoraInicio)
                .AsNoTracking()
                .ToListAsync();

            var dtos = _mapper.Map<IEnumerable<TurnoResponseDto>>(turnos);
            return ApiResponse<IEnumerable<TurnoResponseDto>>.Exito(dtos);
        }

        public async Task<ApiResponse<IEnumerable<object>>> ObtenerHistorialEstadoTurnoAsync(int turnoId)
        {
            var historial = await _context.HistorialEstadoTurnos
                .Where(h => h.TurnoId == turnoId)
                .OrderBy(h => h.FechaCambio)
                .AsNoTracking()
                .Select(h => (object)new
                {
                    h.Id,
                    h.TurnoId,
                    h.EstadoAnteriorId,
                    EstadoAnterior = h.EstadoAnteriorId.HasValue ? ((EstadoTurnoEnum)h.EstadoAnteriorId.Value).ToString() : "N/A",
                    h.EstadoNuevoId,
                    EstadoNuevo = ((EstadoTurnoEnum)h.EstadoNuevoId).ToString(),
                    h.FechaCambio,
                    h.UsuarioId,
                    h.Observaciones
                })
                .ToListAsync();

            return ApiResponse<IEnumerable<object>>.Exito(historial);
        }

        private static object? GetOptionalProp(object target, string[] names)
        {
            var type = target.GetType();
            foreach (var name in names)
            {
                var prop = type.GetProperty(name);
                if (prop != null && prop.CanRead)
                    return prop.GetValue(target);
            }
            return null;
        }

        private static TimeSpan? GetTimeSpanProp(object target, string[] names)
        {
            var val = GetOptionalProp(target, names);
            if (val is TimeSpan ts) return ts;
            if (val is TimeOnly to) return to.ToTimeSpan();
            return null;
        }

        private static void SetOptionalProp(object target, string name, object? value)
        {
            if (value == null) return;
            var prop = target.GetType().GetProperty(name);
            if (prop != null && prop.CanWrite)
            {
                try { prop.SetValue(target, value); } catch { }
            }
        }

using System;
using System.Collections.Generic;
using System.Linq;

    public class VentaService : IVentaService
    {
        private readonly IUnitOfWork _unitOfWork;

        public VentaService(IUnitOfWork unitOfWork)
        {
            _unitOfWork = unitOfWork;
        }

        public async Task<ApiResponse<VentaResponseDto>> ObtenerVentaPorIdAsync(int id, CancellationToken ct = default)
        {

            if (venta == null)
                return ApiResponse<VentaResponseDto>.Fail($"No se encontró la venta con ID {id}.");

            var dto = await MapToResponseDtoAsync(venta, ct);

            var lista = new List<VentaResponseDto>(ventas.Count);

            foreach (var v in ventas)
            {
                lista.Add(await MapToResponseDtoAsync(v, ct));
            }

            return ApiResponse<IReadOnlyList<VentaResponseDto>>.Ok(lista);
        }

        public async Task<ApiResponse<VentaResponseDto>> RegistrarVentaAsync(CrearVentaDto dto, int usuarioId, CancellationToken ct = default)
        {
            var cajaAbierta = await _unitOfWork.SesionesCaja.GetCajaAbiertaActualAsync(ct);
            if (cajaAbierta == null)
                return ApiResponse<VentaResponseDto>.Fail("Debe abrir una sesión de caja antes de registrar ventas.");

            int sesionId = dto.SesionCajaId > 0 ? dto.SesionCajaId : cajaAbierta.Id;

            await _unitOfWork.BeginTransactionAsync(ct);
            try
            {
                var venta = new Venta
                {
                    SesionCajaId = sesionId,
                    ClienteId = dto.ClienteId,
                    TurnoId = dto.TurnoId,
                    UsuarioId = usuarioId,
                    Fecha = DateTime.UtcNow,
                    DetallesVenta = new List<DetalleVenta>(),
                    Pagos = new List<Pago>()
                };

                decimal totalVenta = 0m;

                foreach (var item in dto.Detalles)
                {
                    decimal precioUnitario = 0m;

                    if (item.TipoItem == TipoItemVentaEnum.Producto)
                    {
                        var producto = await _unitOfWork.Productos.GetByIdAsync(item.ItemId);
                        if (producto == null)
                        {
                            await _unitOfWork.RollbackTransactionAsync(ct);
                            return ApiResponse<VentaResponseDto>.Fail($"El producto con ID {item.ItemId} no existe.");
                        }

                        if (producto.StockActual < item.Cantidad)
                        {
                            await _unitOfWork.RollbackTransactionAsync(ct);
                            return ApiResponse<VentaResponseDto>.Fail($"Stock insuficiente para '{producto.Nombre}'. Disponible: {producto.StockActual}.");
                        }

                        precioUnitario = producto.PrecioVenta;
                        producto.StockActual -= item.Cantidad;

                        await _unitOfWork.Productos.RegistrarMovimientoAsync(new MovimientoStock
                        {
                            ProductoId = producto.Id,
                            TipoMovimiento = TipoMovimientoStockEnum.EgresoVentaMostrador,
                            Cantidad = item.Cantidad,
                            Motivo = $"Venta en caja #{sesionId}"
                        });
                    }
                    else
                    {
                        var datosServicio = await _unitOfWork.Ventas.GetDatosServicioAsync(item.ItemId, ct);
                        if (datosServicio == null)
                        {
                            await _unitOfWork.RollbackTransactionAsync(ct);
                            return ApiResponse<VentaResponseDto>.Fail($"El servicio con ID {item.ItemId} no existe.");
                        }

                        precioUnitario = datosServicio.Value.Precio;
                    }

                    decimal subtotalItem = precioUnitario * item.Cantidad;
                    totalVenta += subtotalItem;

                    venta.DetallesVenta.Add(new DetalleVenta
                    {
                        TipoItem = item.TipoItem,
                        ItemId = item.ItemId,
                        Cantidad = item.Cantidad,
                        PrecioUnitario = precioUnitario,
                        Subtotal = subtotalItem,
                        PeluqueroId = item.PeluqueroId
                    });
                }

                venta.Total = totalVenta;

                decimal sumaPagos = dto.Pagos.Sum(p => p.Importe);
                if (Math.Abs(sumaPagos - venta.Total) > 0.01m)
                {
                    await _unitOfWork.RollbackTransactionAsync(ct);
                    return ApiResponse<VentaResponseDto>.Fail($"La suma de los pagos ({sumaPagos:C}) no coincide con el total de la venta ({venta.Total:C}).");
                }

                foreach (var pagoDto in dto.Pagos)
                {
                    venta.Pagos.Add(new Pago
                    {
                        MedioPagoId = pagoDto.MedioPagoId,
                        Importe = pagoDto.Importe
                    });
                }

                await _unitOfWork.Ventas.AddAsync(venta, ct);
                await _unitOfWork.CommitTransactionAsync(ct);

                var ventaCompleta = await _unitOfWork.Ventas.GetByIdCompletaAsync(venta.Id, ct);
                var responseDto = await MapToResponseDtoAsync(ventaCompleta!, ct);

                return ApiResponse<VentaResponseDto>.Ok(responseDto, "Venta registrada exitosamente.");
            }
            catch (Exception ex)
            {
                await _unitOfWork.RollbackTransactionAsync(ct);
                return ApiResponse<VentaResponseDto>.Fail($"Error al procesar la venta: {ex.Message}");
            }
        }

        public async Task<ApiResponse<VentaResponseDto>> AnularVentaAsync(int ventaId, string motivo, int usuarioId, CancellationToken ct = default)
        {
            var venta = await _unitOfWork.Ventas.GetByIdCompletaAsync(ventaId, ct);
            if (venta == null)
                return ApiResponse<VentaResponseDto>.Fail($"No se encontró la venta con ID {ventaId}.");

            if (venta.IsDeleted)
                return ApiResponse<VentaResponseDto>.Fail("La venta ya se encuentra anulada.");

            await _unitOfWork.BeginTransactionAsync(ct);
            try
            {
                foreach (var detalle in venta.DetallesVenta.Where(d => d.TipoItem == TipoItemVentaEnum.Producto))
                {
                    var producto = await _unitOfWork.Productos.GetByIdAsync(detalle.ItemId);
                    if (producto != null)
                    {
                        producto.StockActual += detalle.Cantidad;
                        await _unitOfWork.Productos.RegistrarMovimientoAsync(new MovimientoStock
                        {
                            ProductoId = producto.Id,
                            TipoMovimiento = TipoMovimientoStockEnum.IngresoCompra,
                            Cantidad = detalle.Cantidad,
                            Motivo = $"Restitución por Anulación de Venta #{venta.Id}: {motivo}"
                        });
                    }
                }

                venta.IsDeleted = true;
                venta.DeletedAt = DateTime.UtcNow;
                _unitOfWork.Ventas.Update(venta);

                await _unitOfWork.CommitTransactionAsync(ct);

                var responseDto = await MapToResponseDtoAsync(venta, ct);
                return ApiResponse<VentaResponseDto>.Ok(responseDto, "Venta anulada y stock restituido correctamente.");
            }
            catch (Exception ex)
            {
                await _unitOfWork.RollbackTransactionAsync(ct);
                return ApiResponse<VentaResponseDto>.Fail($"Error al anular la venta: {ex.Message}");
            }
        }

        private async Task<VentaResponseDto> MapToResponseDtoAsync(Venta v, CancellationToken ct)
        {
            var detallesDto = new List<VentaDetalleItemResponseDto>();

            foreach (var d in v.DetallesVenta)
            {
                string conceptoNombre;
                if (d.TipoItem == TipoItemVentaEnum.Producto)
                {
                    var prod = await _unitOfWork.Productos.GetByIdAsync(d.ItemId);
                    conceptoNombre = prod?.Nombre ?? $"Producto #{d.ItemId}";
                }
                else
                {
                    var serv = await _unitOfWork.Ventas.GetDatosServicioAsync(d.ItemId, ct);
                    conceptoNombre = serv?.Nombre ?? $"Servicio #{d.ItemId}";
                }

                string? peluqueroNombre = null;
                if (d.PeluqueroId.HasValue)
                {
                    peluqueroNombre = await _unitOfWork.Ventas.GetNombrePeluqueroAsync(d.PeluqueroId.Value, ct);
                }

                detallesDto.Add(new VentaDetalleItemResponseDto
                {
                    Id = d.Id,
                    TipoItem = d.TipoItem.ToString(),
                    ItemId = d.ItemId,
                    ConceptoNombre = conceptoNombre,
                    Cantidad = d.Cantidad,
                    PrecioUnitario = d.PrecioUnitario,
                    Subtotal = d.Subtotal,
                    PeluqueroId = d.PeluqueroId,
                    PeluqueroNombre = peluqueroNombre
                });
            }

            return new VentaResponseDto
            {
                Id = v.Id,
                SesionCajaId = v.SesionCajaId,
                ClienteId = v.ClienteId,
                ClienteNombre = v.Cliente != null ? $"{v.Cliente.Nombre} {v.Cliente.Apellido}".Trim() : "Consumidor Final",
                TurnoId = v.TurnoId,
                UsuarioId = v.UsuarioId,
                Fecha = v.Fecha,
                Total = v.Total,
                Detalles = detallesDto,
                Pagos = v.Pagos.Select(p => new VentaPagoItemResponseDto
                {
                    Id = p.Id,
                    MedioPagoId = p.MedioPagoId,
                    MedioPagoNombre = p.MedioPago?.Nombre ?? string.Empty,
                    Importe = p.Importe
                }).ToList()
            };
        }

using System.Linq;
using FluentValidation;
using MiPeluqueria.Api.DTOs.Turnos;

namespace MiPeluqueria.Api.Validators.Turnos
{
    public class HorarioLaboralDtoValidator : AbstractValidator<HorarioLaboralDto>
    {
        public HorarioLaboralDtoValidator()
        {
            RuleFor(x => x.DiaSemana)
                .InclusiveBetween(0, 6)
                .WithMessage("El día de la semana debe estar entre 0 (Domingo) y 6 (Sábado).");

            RuleFor(x => x.HoraInicio)
                .LessThan(x => x.HoraFin)
                .When(x => x.Activo)
                .WithMessage("La hora de inicio debe ser anterior a la hora de finalización de la jornada.");

            RuleFor(x => x.HoraInicioDescanso)
                .LessThan(x => x.HoraFinDescanso)
                .When(x => x.HoraInicioDescanso.HasValue && x.HoraFinDescanso.HasValue)
                .WithMessage("La hora de inicio de descanso debe ser anterior a la hora de fin de descanso.");

            RuleFor(x => x.HoraInicioDescanso)
                .GreaterThanOrEqualTo(x => x.HoraInicio)
                .When(x => x.HoraInicioDescanso.HasValue && x.Activo)
                .WithMessage("El descanso no puede comenzar antes del inicio de la jornada laboral.");

            RuleFor(x => x.HoraFinDescanso)
                .LessThanOrEqualTo(x => x.HoraFin)
                .When(x => x.HoraFinDescanso.HasValue && x.Activo)
                .WithMessage("El descanso no puede finalizar después del cierre de la jornada laboral.");
        }
    }

    public class GuardarHorariosPeluqueroDtoValidator : AbstractValidator<GuardarHorariosPeluqueroDto>
    {
        public GuardarHorariosPeluqueroDtoValidator()
        {
            RuleFor(x => x.Horarios)
                .NotNull().WithMessage("La lista de horarios no puede ser nula.");

            RuleForEach(x => x.Horarios)
                .SetValidator(new HorarioLaboralDtoValidator());

            RuleFor(x => x.Horarios)
                .Must(horarios => horarios == null || horarios.Select(h => h.DiaSemana).Distinct().Count() == horarios.Count)
                .WithMessage("No se pueden enviar días de la semana duplicados para el mismo peluquero.");
        }
    }
}
